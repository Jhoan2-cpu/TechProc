-- =====================================================
-- SCHEMA PROCESOS TECNOLÓGICOS - VERSIÓN FINAL
-- PostgreSQL - Tablas Simples
-- =====================================================

-- =====================================================
-- USERS & ROLES
-- =====================================================

CREATE TABLE users (
    id SERIAL PRIMARY KEY,
    first_name VARCHAR(100),
    last_name VARCHAR(100),
    email VARCHAR(255) UNIQUE NOT NULL,
    email_verified_at TIMESTAMP,
    address TEXT,
    birth_date DATE,
    password VARCHAR(255) NOT NULL,
    gender VARCHAR(20),
    country_location VARCHAR(100),
    profile_photo VARCHAR(500),
    role VARCHAR(50) DEFAULT 'student',
    state VARCHAR(50) DEFAULT 'active',
    last_access_ip VARCHAR(45),
    last_access TIMESTAMP,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE instructors (
    instructor_id SERIAL PRIMARY KEY,
    user_id INTEGER NOT NULL,
    bio TEXT,
    expertise_area VARCHAR(255),
    status VARCHAR(50) DEFAULT 'active',
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (user_id) REFERENCES users(id) ON DELETE CASCADE
);

CREATE TABLE students (
    student_id SERIAL PRIMARY KEY,
    user_id INTEGER NOT NULL,
    company_id INTEGER,
    status VARCHAR(50) DEFAULT 'active',
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (user_id) REFERENCES users(id) ON DELETE CASCADE
);

CREATE TABLE employees (
    id_employee SERIAL PRIMARY KEY,
    name VARCHAR(100),
    user_id INTEGER NOT NULL,
    employee_role VARCHAR(50),
    status VARCHAR(50) DEFAULT 'active',
    schedule TEXT,
    speciality VARCHAR(255),
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (user_id) REFERENCES users(id) ON DELETE CASCADE
);

-- =====================================================
-- COURSES & CATEGORIES
-- =====================================================

CREATE TABLE categories (
    category_id SERIAL PRIMARY KEY,
    name VARCHAR(100) NOT NULL,
    slug VARCHAR(100) UNIQUE,
    image VARCHAR(255),
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE courses (
    course_id SERIAL PRIMARY KEY,
    title VARCHAR(255) NOT NULL,
    description TEXT,
    course_image VARCHAR(255),
    video_url VARCHAR(255),
    duration NUMERIC(8,2),
    sessions INTEGER,
    selling_price NUMERIC(10,2),
    discount_price NUMERIC(10,2),
    prerequisites TEXT,
    certificate_name BOOLEAN DEFAULT FALSE,
    certificate_issuer VARCHAR(255),
    bestseller BOOLEAN DEFAULT FALSE,
    featured BOOLEAN DEFAULT FALSE,
    highest_rated BOOLEAN DEFAULT FALSE,
    status BOOLEAN DEFAULT TRUE,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE course_categories (
    id_course_catg SERIAL PRIMARY KEY,
    course_id INTEGER NOT NULL,
    category_id INTEGER NOT NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (course_id) REFERENCES courses(course_id) ON DELETE CASCADE,
    FOREIGN KEY (category_id) REFERENCES categories(category_id) ON DELETE CASCADE
);

CREATE TABLE course_instructors (
    id_courseinst SERIAL PRIMARY KEY,
    instructor_id INTEGER NOT NULL,
    course_id INTEGER NOT NULL,
    assigned_date TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (instructor_id) REFERENCES instructors(instructor_id) ON DELETE CASCADE,
    FOREIGN KEY (course_id) REFERENCES courses(course_id) ON DELETE CASCADE
);

CREATE TABLE course_contents (
    id SERIAL PRIMARY KEY,
    course_id INTEGER NOT NULL,
    session INTEGER,
    type VARCHAR(50),
    title VARCHAR(255),
    content TEXT,
    order_number INTEGER,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (course_id) REFERENCES courses(course_id) ON DELETE CASCADE
);

-- =====================================================
-- ENROLLMENT (MATRÍCULAS)
-- =====================================================

CREATE TABLE academic_periods (
    academic_period_id SERIAL PRIMARY KEY,
    name VARCHAR(255) NOT NULL,
    start_date DATE NOT NULL,
    end_date DATE NOT NULL,
    status VARCHAR(50) DEFAULT 'open',
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE course_offerings (
    course_offering_id SERIAL PRIMARY KEY,
    course_id INTEGER NOT NULL,
    academic_period_id INTEGER NOT NULL,
    instructor_id INTEGER,
    schedule TEXT,
    delivery_method VARCHAR(50) DEFAULT 'regular',
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (course_id) REFERENCES courses(course_id),
    FOREIGN KEY (academic_period_id) REFERENCES academic_periods(academic_period_id),
    FOREIGN KEY (instructor_id) REFERENCES instructors(instructor_id)
);

CREATE TABLE enrollment (
    enrollment_id SERIAL PRIMARY KEY,
    student_id INTEGER NOT NULL,
    academic_period_id INTEGER NOT NULL,
    enrollment_date TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    status VARCHAR(50) DEFAULT 'active',
    enrollment_type VARCHAR(50) DEFAULT 'new',
    FOREIGN KEY (student_id) REFERENCES students(student_id),
    FOREIGN KEY (academic_period_id) REFERENCES academic_periods(academic_period_id)
);

CREATE TABLE enrollment_details (
    enrollment_detail_id SERIAL PRIMARY KEY,
    enrollment_id INTEGER NOT NULL,
    course_offering_id INTEGER NOT NULL,
    status VARCHAR(50) DEFAULT 'active',
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (enrollment_id) REFERENCES enrollment(enrollment_id) ON DELETE CASCADE,
    FOREIGN KEY (course_offering_id) REFERENCES course_offerings(course_offering_id)
);

-- =====================================================
-- TICKETS & ESCALATIONS
-- =====================================================

CREATE TABLE ticket (
    ticket_id SERIAL PRIMARY KEY,
    assigned_technician INTEGER,
    user_id INTEGER NOT NULL,
    title VARCHAR(255) NOT NULL,
    description TEXT NOT NULL,
    priority VARCHAR(50) DEFAULT 'media',
    status VARCHAR(50) DEFAULT 'abierto',
    creation_date TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    assignment_date TIMESTAMP,
    resolution_date TIMESTAMP,
    close_date TIMESTAMP,
    category VARCHAR(100),
    notes TEXT,
    FOREIGN KEY (assigned_technician) REFERENCES employees(id_employee),
    FOREIGN KEY (user_id) REFERENCES users(id)
);

CREATE TABLE ticket_tracking (
    ticket_tracking_id SERIAL PRIMARY KEY,
    ticket_id INTEGER NOT NULL,
    comment TEXT,
    action_type VARCHAR(50),
    follow_up_date TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (ticket_id) REFERENCES ticket(ticket_id) ON DELETE CASCADE
);

CREATE TABLE escalations (
    escalation_id SERIAL PRIMARY KEY,
    ticket_id INTEGER NOT NULL,
    technician_origin_id INTEGER,
    technician_destiny_id INTEGER,
    escalation_reason VARCHAR(255),
    observations TEXT,
    escalation_date TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    approved BOOLEAN DEFAULT FALSE,
    FOREIGN KEY (ticket_id) REFERENCES ticket(ticket_id) ON DELETE CASCADE,
    FOREIGN KEY (technician_origin_id) REFERENCES employees(id_employee),
    FOREIGN KEY (technician_destiny_id) REFERENCES employees(id_employee)
);

-- =====================================================
-- SECURITY
-- =====================================================

CREATE TABLE security_logs (
    id_security_log SERIAL PRIMARY KEY,
    user_id INTEGER,
    event_type VARCHAR(100) NOT NULL,
    description TEXT,
    source_ip VARCHAR(45),
    event_date TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (user_id) REFERENCES users(id)
);

CREATE TABLE blocked_ips (
    id_blocked_ip SERIAL PRIMARY KEY,
    ip_address VARCHAR(45) NOT NULL,
    reason VARCHAR(255),
    block_date TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    active BOOLEAN DEFAULT TRUE
);

CREATE TABLE security_alerts (
    id_security_alert SERIAL PRIMARY KEY,
    threat_type VARCHAR(100) NOT NULL,
    severity VARCHAR(50) DEFAULT 'medium',
    status VARCHAR(50) DEFAULT 'new',
    blocked_ip_id INTEGER,
    detection_date TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (blocked_ip_id) REFERENCES blocked_ips(id_blocked_ip)
);

CREATE TABLE incidents (
    id_incident SERIAL PRIMARY KEY,
    alert_id INTEGER,
    responsible_id INTEGER,
    title VARCHAR(255) NOT NULL,
    status VARCHAR(50) DEFAULT 'open',
    report_date TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (alert_id) REFERENCES security_alerts(id_security_alert),
    FOREIGN KEY (responsible_id) REFERENCES employees(id_employee)
);

CREATE TABLE active_sessions (
    session_id SERIAL PRIMARY KEY,
    user_id INTEGER NOT NULL,
    ip_address VARCHAR(45),
    device VARCHAR(255),
    start_date TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    active BOOLEAN DEFAULT TRUE,
    blocked BOOLEAN DEFAULT FALSE,
    FOREIGN KEY (user_id) REFERENCES users(id) ON DELETE CASCADE
);

CREATE TABLE security_configuration (
    id_security_configuration SERIAL PRIMARY KEY,
    user_id INTEGER,
    modulo VARCHAR(100),
    parameter VARCHAR(100) NOT NULL,
    value TEXT,
    active BOOLEAN DEFAULT TRUE,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (user_id) REFERENCES users(id)
);

-- =====================================================
-- SOFTWARE & LICENSES
-- =====================================================

CREATE TABLE licenses (
    id_license SERIAL PRIMARY KEY,
    software_name VARCHAR(255) NOT NULL,
    license_key TEXT,
    license_type VARCHAR(100),
    provider VARCHAR(255),
    purchase_date DATE,
    expiration_date DATE,
    seats_total INTEGER,
    seats_used INTEGER DEFAULT 0,
    cost_annual NUMERIC(12,2),
    status VARCHAR(50) DEFAULT 'active',
    responsible_id INTEGER,
    notes TEXT,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (responsible_id) REFERENCES employees(id_employee)
);

CREATE TABLE softwares (
    id_software SERIAL PRIMARY KEY,
    software_name VARCHAR(255) NOT NULL,
    version VARCHAR(100),
    category VARCHAR(100),
    vendor VARCHAR(255),
    license_id INTEGER,
    installation_date TIMESTAMP,
    last_update TIMESTAMP,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (license_id) REFERENCES licenses(id_license)
);

-- =====================================================
-- CHATBOTS
-- =====================================================

CREATE TABLE chatbot_conversations (
    id_conversation SERIAL PRIMARY KEY,
    started_date TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    ended_date TIMESTAMP,
    satisfaction_rating INTEGER CHECK (satisfaction_rating BETWEEN 1 AND 5),
    feedback TEXT,
    resolved BOOLEAN DEFAULT FALSE,
    handed_to_human BOOLEAN DEFAULT FALSE
);

CREATE TABLE chatbot_faq (
    id_faq SERIAL PRIMARY KEY,
    question TEXT NOT NULL,
    answer TEXT NOT NULL,
    category VARCHAR(100),
    keywords JSONB,
    active BOOLEAN DEFAULT TRUE,
    usage_count INTEGER DEFAULT 0,
    created_date TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_date TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE chatbot_messages (
    id_message SERIAL PRIMARY KEY,
    conversation_id INTEGER NOT NULL,
    sender VARCHAR(50) NOT NULL,
    message TEXT NOT NULL,
    timestamp TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    faq_matched INTEGER,
    FOREIGN KEY (conversation_id) REFERENCES chatbot_conversations(id_conversation) ON DELETE CASCADE,
    FOREIGN KEY (faq_matched) REFERENCES chatbot_faq(id_faq)
);

-- =====================================================
-- NEWS & ANNOUNCEMENTS
-- =====================================================

CREATE TABLE news (
    id_news SERIAL PRIMARY KEY,
    title VARCHAR(255) NOT NULL,
    slug VARCHAR(255) UNIQUE,
    summary TEXT,
    content TEXT,
    featured_image VARCHAR(500),
    author_id INTEGER,
    category VARCHAR(100),
    tags JSONB,
    status VARCHAR(50) DEFAULT 'draft',
    views INTEGER DEFAULT 0,
    published_date TIMESTAMP,
    created_date TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_date TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    seo_title VARCHAR(255),
    seo_description TEXT,
    FOREIGN KEY (author_id) REFERENCES users(id)
);

CREATE TABLE announcements (
    id_announcement SERIAL PRIMARY KEY,
    title VARCHAR(255) NOT NULL,
    content TEXT NOT NULL,
    image_url VARCHAR(500),
    display_type VARCHAR(50),
    target_page VARCHAR(100),
    link_url VARCHAR(500),
    button_text VARCHAR(100),
    status VARCHAR(50) DEFAULT 'draft',
    start_date TIMESTAMP,
    end_date TIMESTAMP,
    views INTEGER DEFAULT 0,
    created_by INTEGER,
    created_date TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (created_by) REFERENCES users(id)
);

CREATE TABLE alerts (
    id_alert SERIAL PRIMARY KEY,
    message TEXT NOT NULL,
    type VARCHAR(50),
    status VARCHAR(50) DEFAULT 'active',
    link_url VARCHAR(500),
    link_text VARCHAR(100),
    start_date TIMESTAMP,
    end_date TIMESTAMP,
    priority INTEGER DEFAULT 1,
    created_by INTEGER,
    created_date TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (created_by) REFERENCES users(id)
);

-- =====================================================
-- CONTACT FORMS
-- =====================================================

CREATE TABLE contact_forms (
    id_contact SERIAL PRIMARY KEY,
    full_name VARCHAR(255) NOT NULL,
    email VARCHAR(255) NOT NULL,
    phone VARCHAR(20),
    company VARCHAR(255),
    subject VARCHAR(255),
    message TEXT NOT NULL,
    form_type VARCHAR(50),
    status VARCHAR(50) DEFAULT 'pending',
    assigned_to INTEGER,
    response TEXT,
    response_date TIMESTAMP,
    submission_date TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (assigned_to) REFERENCES employees(id_employee)
);

CREATE INDEX idx_users_email ON users(email);
CREATE INDEX idx_users_role ON users(role);

CREATE INDEX idx_courses_title ON courses(title);
CREATE INDEX idx_courses_status ON courses(status);

-- Tickets
CREATE INDEX idx_ticket_status ON ticket(status);
CREATE INDEX idx_ticket_priority ON ticket(priority);
CREATE INDEX idx_ticket_user ON ticket(user_id);

-- Security
CREATE INDEX idx_security_logs_user ON security_logs(user_id);
CREATE INDEX idx_security_logs_date ON security_logs(event_date);
CREATE INDEX idx_security_alerts_severity ON security_alerts(severity);

-- Chatbot
CREATE INDEX idx_chatbot_messages_conversation ON chatbot_messages(conversation_id);
CREATE INDEX idx_chatbot_faq_category ON chatbot_faq(category);

-- News
CREATE INDEX idx_news_slug ON news(slug);
CREATE INDEX idx_news_status ON news(status);
